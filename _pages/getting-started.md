<script>
function toggleExtra(button) {
  const tile = button.closest('.service-tile');
  const extraItems = tile.querySelectorAll('.tile-extra-item');
  const extraText = tile.querySelector('.tile-extra-text');

  // Check if items are currently hidden
  const isHidden = extraItems[0] && extraItems[0].style.display === 'none';

  // Toggle list items
  extraItems.forEach(item => {
    item.style.display = isHidden ? 'list-item' : 'none';
  });

  // Toggle extra text if it exists
  if (extraText) {
    extraText.style.display = isHidden ? 'block' : 'none';
  }

  // Update button text and ARIA attributes
  button.textContent = isHidden ? 'Show Less ▲' : 'Show More ▼';
  button.setAttribute('aria-expanded', isHidden ? 'true' : 'false');
}

// Initialize ARIA attributes on page load
document.addEventListener('DOMContentLoaded', function() {
  document.querySelectorAll('.tile-toggle').forEach(btn => {
    btn.setAttribute('aria-expanded', 'false');
    btn.setAttribute('aria-label', 'Toggle additional service details');
  });
});
</script>
